cmake_minimum_required(VERSION 3.20)
project(exact_generation_upgrade_consumer LANGUAGES CXX)
find_package(agent_memory CONFIG REQUIRED)
option(UPGRADE_NEW "Build the NEW installed consumer" OFF)
add_executable(exact_generation_upgrade consumer.cpp)
set_target_properties(exact_generation_upgrade PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_link_libraries(exact_generation_upgrade PRIVATE agent_memory::agent_memory)
if(UPGRADE_NEW)
    target_compile_definitions(exact_generation_upgrade PRIVATE UPGRADE_NEW=1)
endif()
